"""SOURCE lifecycle proof: stop only the actual owned child; never a platform PID."""
import asyncio,json,os,signal,subprocess,time
from pathlib import Path
import pytest
from test_trading_api import client,_env
import test_stateful_link_binding as owner

@pytest.mark.asyncio
@pytest.mark.parametrize('ending',['cancel','timeout'])
async def test_actual_owner_child_is_reaped_on_cancel_or_original_timeout(client,tmp_path,monkeypatch,ending):
    real_popen=subprocess.Popen;children=[];spawned=asyncio.Event();ticks=[]
    def capture(*args,**kwargs):
        child=real_popen(*args,**kwargs)
        command=args[0] if args else kwargs.get('args',[])
        if isinstance(command,list) and len(command)>2 and command[1]=='-c' and 'exit_after_claim' in command[2]:
            assert kwargs['start_new_session'] is True and os.getpgid(child.pid)==child.pid
            children.append(child)
            os.kill(child.pid,signal.SIGSTOP)
            spawned.set()
        return child
    monkeypatch.setattr(subprocess,'Popen',capture)
    async def tick():
        while True:
            await asyncio.sleep(.05);ticks.append(time.monotonic())
    pulse=asyncio.create_task(tick());task=asyncio.create_task(owner.test_real_link_process_death_keeps_bound_original_unknown_without_native_resend(client,tmp_path,monkeypatch));started=time.monotonic()
    try:
        await asyncio.wait_for(spawned.wait(),timeout=15)
        if ending=='cancel':
            await asyncio.sleep(.1);task.cancel()
            with pytest.raises(asyncio.CancelledError):await asyncio.wait_for(task,timeout=10)
        else:
            with pytest.raises(subprocess.TimeoutExpired):await asyncio.wait_for(task,timeout=45)
        assert task.done() and len(children)==1 and children[0].returncode==-signal.SIGKILL
        with pytest.raises(ChildProcessError):os.waitpid(children[0].pid,os.WNOHANG)
        leaked=[]
        for other in asyncio.all_tasks():
            coro=other.get_coro();frame=getattr(coro,'cr_frame',None)
            func=frame.f_locals.get('func') if frame else None
            if getattr(func,'__self__',None) is children[0]:leaked.append(other.get_name())
        assert not leaked and len(ticks)>0
        (tmp_path/'lifecycle-diagnostic.json').write_text(json.dumps({'ending':ending,'actual_child_pid':children[0].pid,'returncode':children[0].returncode,'reaped':True,'owner_wait_tasks_remaining':leaked,'parent_loop_ticks':len(ticks),'seconds':time.monotonic()-started,'original_timeout_seconds':30,'scope':'Actual source owner test+actual unchanged childscript; external fixture SIGSTOP only its just-created ownsession child to exercisecancel/timeout. No platformsignals/DBtimeout/assertion/authoritychanges.'},indent=2)+'\n')
    finally:
        if not task.done():
            task.cancel()
            await asyncio.gather(task,return_exceptions=True)
        pulse.cancel();await asyncio.gather(pulse,return_exceptions=True)
        for child in children:
            if child.poll() is None:os.killpg(child.pid,signal.SIGKILL)
            child.communicate(timeout=5)
