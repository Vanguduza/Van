def command(args,stdin=None,timeout=60):
 return subprocess.run(args,input=stdin,text=True,capture_output=True,timeout=timeout)
def qi(name):
 if not re.fullmatch('[a-z][a-z0-9_]*',name):raise ValueError('ROLE_IDENTIFIER')
 return '"'+name+'"'
def call(args,stdin=None,timeout=60):
 p=command(args,stdin,timeout)
 if p.returncode:raise RuntimeError(p.stderr[-2500:])
 return p.stdout

  check('fixture_only_new_data_mounted',len(settings['Mounts'])==1 and settings['Mounts'][0]['Source']==str(state))
  for _ in range(100):
