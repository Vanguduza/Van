cmake_minimum_required(VERSION 3.22)
project(van_pinned_rive_native_compatibility LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT RIVE_SOURCE OR NOT CAIRO_INCLUDE_DIR OR NOT CAIRO_LIBRARY)
    message(FATAL_ERROR "Explicit verified RIVE_SOURCE, CAIRO_INCLUDE_DIR and CAIRO_LIBRARY required")
endif()
file(GLOB_RECURSE CORE_SOURCE CONFIGURE_DEPENDS "${RIVE_SOURCE}/src/*.cpp")
list(LENGTH CORE_SOURCE CORE_COUNT)
if(NOT CORE_COUNT EQUAL 468)
    message(FATAL_ERROR "The pinned native source set must contain exactly 468 original .cpp files")
endif()
# Compile pristine source only. Do not rewrite the upstream runtime to fit a rig.
# Optional text/audio/flex-layout support is deliberately outside this diagnostic.
add_library(rive_pinned_core STATIC ${CORE_SOURCE})
target_include_directories(rive_pinned_core PUBLIC "${RIVE_SOURCE}/include" "${RIVE_SOURCE}")
target_compile_definitions(rive_pinned_core PRIVATE _RIVE_INTERNAL_)
target_compile_options(rive_pinned_core PRIVATE -O1 -g0 -fno-omit-frame-pointer)
add_executable(native_probe native_probe.cpp)
target_include_directories(native_probe PRIVATE "${CAIRO_INCLUDE_DIR}")
target_compile_options(native_probe PRIVATE -O1 -g0)
target_link_libraries(native_probe PRIVATE rive_pinned_core "${CAIRO_LIBRARY}" pthread dl)
